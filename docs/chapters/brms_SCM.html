33  SCM: Bayesian

33.1 Overview

This short tutorial shows the same SCM/g-computation logic two ways:

  1. A frequentist mgcv::gam() workflow using bootstrap draws. This is the same SCM workflow that you saw in the previous tutorial.
  2. A Bayesian brms workflow using posterior draws.

The goal is not to introduce every feature of brms, but rather to show how total, controlled direct, and indirect effects can be estimated in a way that parallels the frequentist GAM approach. Let us go through the steps.

33.3 2. Frequentist GAM bootstrap comparison

The frequentist comparison uses the same structural equations as the Bayesian version that we will use later in this tutorial. As you recall, the bootstrap gives a sampling distribution of the total, controlled direct, and indirect effects.

# we are setting the random-effect levels as the most common levels for 'site' and 'observer'
mode_factor <- function(x) {
  factor(names(which.max(table(x))), levels = levels(x))
}

Next, we can use a function that fits all of the parent-set models; this fits the entire SCM. This is akin to the classical Structural Equation Modeling (SEM) approach, except that we do not alter the covariates in the submodels (as many practitioners of SEM –even piecewise-SEM– do).

fit_gam_scm <- function(d) {
  d <- d |>
    mutate(
      site = factor(site, levels = levels(dat$site)),
      obs  = factor(obs,  levels = levels(dat$obs))
    )

  list(
    m1 = gam(
      M1 ~ s(X) + s(C) + s(site, bs = "re") + s(obs, bs = "re"),
      data = d,
      family = gaussian(),
      method = "REML"
    ),
    m2 = gam(
      M2 ~ s(X) + s(M1) + s(site, bs = "re") + s(obs, bs = "re"),
      data = d,
      family = gaussian(),
      method = "REML"
    ),
    y = gam(
      Y ~ s(X) + s(M2) + s(C) + s(site, bs = "re") + s(obs, bs = "re"),
      data = d,
      family = scat(),
      method = "REML"
    )
  )
}

Then, we can create another function to help us with our explicit intervention (i.e. a “natural” prediction). Recall that “intervention” in this case means the specification of values of our causal variable of interest (X). Note the arguments for x0 and x1 in this function.

gam_gcomp_one <- function(d, x0, x1, draw = 1) {
  fits <- fit_gam_scm(d)

  base_dat <- tibble(
    C = dat$C,
    site = mode_factor(dat$site),
    obs  = mode_factor(dat$obs)
  )

  predict_do_x <- function(x_value) {
    d_x <- base_dat |> mutate(X = x_value)

    m1_hat <- predict(fits$m1, newdata = d_x, type = "response")
    m2_hat <- predict(fits$m2, newdata = d_x |> mutate(M1 = m1_hat), type = "response")
    y_hat  <- predict(fits$y,  newdata = d_x |> mutate(M1 = m1_hat, M2 = m2_hat), type = "response")

    mean(y_hat)
  }

  predict_cde <- function(x_treat, x_mediator_ref) {
    d_ref <- base_dat |> mutate(X = x_mediator_ref)

    m1_ref <- predict(fits$m1, newdata = d_ref, type = "response")
    m2_ref <- predict(fits$m2, newdata = d_ref |> mutate(M1 = m1_ref), type = "response")

    y_hat <- predict(
      fits$y,
      newdata = base_dat |> mutate(X = x_treat, M2 = m2_ref),
      type = "response"
    )

    mean(y_hat)
  }

  total_effect <- predict_do_x(x1) - predict_do_x(x0)
  direct_effect <- predict_cde(x1, x0) - predict_cde(x0, x0)
  indirect_effect <- total_effect - direct_effect

  tibble(
    .draw = draw,
    method = "GAM bootstrap",
    effect = c("Total effect", "Controlled direct effect", "Indirect effect"),
    estimate = c(total_effect, direct_effect, indirect_effect)
  )
}

set.seed(20260412)

B <- 200

boot_ids <- replicate(
  B,
  sample(seq_len(nrow(dat)), replace = TRUE),
  simplify = FALSE
)

gam_effects <- purrr::map2_dfr(
  boot_ids,
  seq_along(boot_ids),
  \(idx, draw) gam_gcomp_one(dat[idx, ], x0 = x0, x1 = x1, draw = draw)
)

33.4 3. Fit the Bayesian DAG path models

These are the same final structural equations used in the frequentist GAM workflow from the previous tutorial), translated directly into brms syntax.

NoteChoosing priors

For an example like this one, we can use weak priors that keep estimates in ecologically plausible ranges without forcing the answer (i.e. the final effect). A good and simple starting point is normal(0, 1) for slopes. Predictors are on a reasonable scale, but we may wish to allow a slightly wider prior for intercepts. By convention, we can use exponential priors for standard deviations and smooth penalties. For scaled-t distributions (outcomes), we can place a prior on nu so the model can represent heavier tails without becoming completely unconstrained.

First, we set our priors for the Bayesian model.

priors_gaussian <- c(
  prior(normal(0, 1), class = "b"),
  prior(normal(0, 2), class = "Intercept"),
  prior(exponential(1), class = "sd"),
  prior(exponential(1), class = "sds"),
  prior(exponential(1), class = "sigma")
)

priors_student <- c(
  prior(normal(0, 1), class = "b"),
  prior(normal(0, 2), class = "Intercept"),
  prior(exponential(1), class = "sd"),
  prior(exponential(1), class = "sds"),
  prior(exponential(1), class = "sigma"),
  prior(gamma(2, 0.1), class = "nu")
)

What do these mean? Work through them:

  • b ~ normal(0, 1) The effects (beta coefficients) are likely small and centered near zero
  • Intercept ~ normal(0, 2) The outcome starts near zero but can vary more
  • sd, sds, sigma ~ exponential(1) Variability is positive and usually not huge (regularizing noise and random effects)
  • nu ~ gamma(2, 0.1) (Scaled t-distribution model only) Allows heavier tails (outliers), but keeps them reasonable

In plain terms, what do these mean? Collectively, yhey mean this: Most effects are relatively small (centered around 0); Noise is minimized (low measure of variance), and the model allows some outliers without going too wild! Priors are just used to help each model finds some meaningful values.

Now, let us fit the individual path-specific models using brms (same formulation as mgcv::gam):

fit_m1 <- brm(
  M1 ~ s(X) + s(C) + (1 | site) + (1 | obs),
  data = dat,
  family = gaussian(),
  prior = priors_gaussian,
  chains = 4, cores = 4, iter = 2000,
  seed = 20260408,
  silent = 2, refresh = 0
)

fit_m2 <- brm(
  M2 ~ s(X) + s(M1) + (1 | site) + (1 | obs),
  data = dat,
  family = gaussian(),
  prior = priors_gaussian,
  chains = 4, cores = 4, iter = 2000,
  seed = 20260409,
  silent = 2, refresh = 0
)

fit_y <- brm(
  Y ~ s(X) + s(M2) + s(C) + (1 | site) + (1 | obs),
  data = dat,
  family = student(),
  prior = priors_student,
  chains = 4, cores = 4, iter = 2000,
  seed = 20260410,
  silent = 2, refresh = 0
)

At this point, you can view the output of each model. Due to the computational intensiveness of the program Stan (for which brms is the wrapper), it is generally a good idea to test your model structure using mgcv::gam or some other frequentist approach. You will see many familiar output components using the summary() function on one of these brms model objects, but these are not shown here.

33.5 4. Bayesian posterior g-computation

For each posterior draw, we can predict the mediator values under the chosen intervention and then feed those predictions into the next model in the causal chain. This is the Bayesian analogue of the bootstrap g-computation above. It is very analogous, as you see.

WarningWARNING!! This code is very computationally intensive. Depending on your computer’s configuration, it can take a very long time. 
new_base <- tibble(
  C = dat$C,
  site = mode_factor(dat$site),
  obs  = mode_factor(dat$obs)
)

posterior_mean_response <- function(fit, newdata, draw_id) {
  as.vector(posterior_epred(
    fit,
    newdata = newdata,
    draw_ids = draw_id,
    re_formula = NULL,
    allow_new_levels = FALSE
  ))
}

set.seed(20260411)
draw_ids <- sample(seq_len(ndraws(fit_y)), size = min(500, ndraws(fit_y)))

predict_do_x_one <- function(draw_id, x_value) {
  dat_x <- new_base |> mutate(X = x_value)
  m1 <- posterior_mean_response(fit_m1, dat_x, draw_id)
  m2 <- posterior_mean_response(fit_m2, dat_x |> mutate(M1 = m1), draw_id)
  y  <- posterior_mean_response(fit_y,  dat_x |> mutate(M1 = m1, M2 = m2), draw_id)

  tibble(.draw = draw_id, ybar = mean(y))
}

predict_cde_one <- function(draw_id, x_treat, x_mediator_ref) {
  dat_ref <- new_base |> mutate(X = x_mediator_ref)
  m1_ref <- posterior_mean_response(fit_m1, dat_ref, draw_id)
  m2_ref <- posterior_mean_response(fit_m2, dat_ref |> mutate(M1 = m1_ref), draw_id)

  y <- posterior_mean_response(
    fit_y,
    new_base |> mutate(X = x_treat, M2 = m2_ref),
    draw_id
  )

  tibble(.draw = draw_id, ybar = mean(y))
}

predict_do_x <- function(x_value) {
  purrr::map_dfr(draw_ids, predict_do_x_one, x_value = x_value)
}

predict_cde <- function(x_treat, x_mediator_ref) {
  purrr::map_dfr(
    draw_ids,
    predict_cde_one,
    x_treat = x_treat,
    x_mediator_ref = x_mediator_ref
  )
}

post_te <- predict_do_x(x1) |>
  rename(y_x1 = ybar) |>
  left_join(predict_do_x(x0) |> rename(y_x0 = ybar), by = ".draw") |>
  transmute(.draw, total_effect = y_x1 - y_x0)

post_cde <- predict_cde(x1, x0) |>
  rename(y_x1_m0 = ybar) |>
  left_join(predict_cde(x0, x0) |> rename(y_x0_m0 = ybar), by = ".draw") |>
  transmute(.draw, direct_effect = y_x1_m0 - y_x0_m0)

post_effects <- post_te |>
  left_join(post_cde, by = ".draw") |>
  mutate(indirect_effect = total_effect - direct_effect) |>
  pivot_longer(-.draw, names_to = "effect", values_to = "estimate") |>
  mutate(
    method = "brms posterior",
    effect = recode(
      effect,
      total_effect = "Total effect",
      direct_effect = "Controlled direct effect",
      indirect_effect = "Indirect effect"
    )
  ) |>
  select(.draw, method, effect, estimate)

33.6 5. Compare GAM bootstrap and brms posterior effects

The table and plot below compare the frequentist bootstrap distribution to the Bayesian posterior distribution. They are directly comparable because both are stored with the same columns: .draw, method, effect, and estimate.

comparison_effects <- bind_rows(gam_effects, post_effects)

effect_summary <- comparison_effects |>
  group_by(method, effect) |>
  median_qi(estimate, .width = 0.95) |>
  transmute(
    method,
    effect,
    estimate = estimate,
    lwr_95 = .lower,
    upr_95 = .upper
  ) |>
  arrange(effect, method)
